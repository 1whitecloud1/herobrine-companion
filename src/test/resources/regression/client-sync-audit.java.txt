import java.util.*;
import java.util.concurrent.CompletableFuture;
import java.util.function.Supplier;

public class ClientSyncAuditProbe {
    private static int checks;
    public static void verify() {
        rewardSnapshotsAndButtons();
        rewardAcknowledgements();
        fullQuestCopy();
        connectionCleanup();
        liveQuestCooldown();
        System.out.println("ClientSyncAuditChecks: " + checks + " checks passed");
    }

    private static void rewardSnapshotsAndButtons() {
        HeroEntity hero = new HeroEntity(7, new World());
        Minecraft.getInstance().level = hero.world;
        hero.world.entities.put(7, hero);
        HeroRewardScreen screen = new HeroRewardScreen(hero);
        check(!screen.needsStateRefresh(), "An unchanged reward screen does not rebuild");
        ClientStateSync.applySyncRewards(7, Set.of(1, 2));
        check(hero.hasClaimedReward(1) && hero.hasClaimedReward(2) && screen.needsStateRefresh(), "An incoming snapshot refreshes the reward buttons");
        screen.capture();
        ClientStateSync.applySyncRewards(7, Set.of(2));
        check(!hero.hasClaimedReward(1) && hero.hasClaimedReward(2), "A removed claim is removed from the client mirror");
        ClientStateSync.applySyncRewards(7, Set.of());
        check(!hero.hasClaimedReward(2), "An empty server snapshot clears all old client claims");
        screen.capture();
        hero.trust = 50;
        check(screen.needsStateRefresh(), "Trust changes refresh locked/unlocked buttons without reopening the screen");
        screen.capture();
        ClientHeroContent.revision++;
        check(screen.needsStateRefresh(), "Content reload still refreshes reward definitions");
        screen.capture();
        HeroRewards.Reward reward = new HeroRewards.Reward(1, 50);
        Button pending = new Button();
        PacketHandler.requests.clear();
        screen.requestReward(reward, pending);
        check(PacketHandler.requests.size() == 1 && !pending.active && pending.label.equals("claim"),
                "Sending a claim waits for server acknowledgement without pretending it was granted");
        ClientStateSync.applySyncRewards(7, Set.of());
        check(screen.needsStateRefresh() && !hero.hasClaimedReward(1), "A rejected claim reply refreshes the pending button even when the set is unchanged");
        screen.capture();
        hero.trust = 0;
        PacketHandler.requests.clear();
        screen.requestReward(reward, new Button());
        check(PacketHandler.requests.isEmpty(), "The request callback rechecks current trust");
        hero.trust = 50;
        ClientStateSync.applySyncRewards(7, Set.of(1));
        screen.requestReward(reward, new Button());
        check(PacketHandler.requests.isEmpty(), "The request callback rechecks current claimed state");
        hero.world.entities.remove(7);
        check(screen.needsStateRefresh(), "Entity disappearance cannot leave old buttons enabled");
    }

    private static void rewardAcknowledgements() {
        ServerLevel level = new ServerLevel();
        HeroEntity hero = new HeroEntity(9, level);
        level.entities.put(9, hero);
        ServerPlayer player = new ServerPlayer(level);
        HeroRewards.definitions.put(1, new HeroRewards.Reward(1, 50));
        hero.trust = 0;
        PacketHandler.replies.clear();
        PacketHandler.broadcasts.clear();
        ClaimRewardPacket.invoke(player, 9, 1);
        check(player.inventory.items == 0 && PacketHandler.replies.size() == 1 && PacketHandler.broadcasts.isEmpty(),
                "Insufficient trust returns authoritative state without granting or broadcasting a claim");
        check(PacketHandler.replies.get(0).claimedRewards().isEmpty(), "A denied request cannot mark its reward claimed");
        hero.trust = 50;
        PacketHandler.replies.clear();
        ClaimRewardPacket.invoke(player, 9, 1);
        check(player.inventory.items == 2 && hero.hasClaimedReward(1), "An eligible claim grants its actual items once");
        check(PacketHandler.replies.size() == 1 && PacketHandler.broadcasts.size() == 1,
                "The claimant and other tracking viewers receive the new claimed state");
        check(PacketHandler.replies.get(0).claimedRewards().equals(Set.of(1)), "The acknowledgement contains the canonical server snapshot");
        PacketHandler.replies.clear();
        PacketHandler.broadcasts.clear();
        ClaimRewardPacket.invoke(player, 9, 1);
        check(player.inventory.items == 2 && PacketHandler.replies.size() == 1 && PacketHandler.broadcasts.isEmpty(),
                "A duplicate request gets a reply without duplicate items or redundant observer broadcasts");
        PacketHandler.replies.clear();
        ClaimRewardPacket.invoke(player, 9, 999);
        check(PacketHandler.replies.size() == 1 && PacketHandler.replies.get(0).claimedRewards().equals(Set.of(1)),
                "An obsolete reward definition still receives an accurate reply");
    }

    private static void fullQuestCopy() {
        String[] keys = {"HeroActiveQuestId", "HeroActiveQuestProgress", "HeroQuestTargetUUID", "HeroQuestHeroUUID",
                "HeroQuestPositions", "HeroQuestFlags", "HeroQuestCooldowns"};
        CompoundTag original = new CompoundTag(), replacement = new CompoundTag();
        replacement.put("Unrelated", new ValueTag("keep"));
        for (String key : keys) {
            CompoundTag nested = new CompoundTag();
            nested.put("value", new ValueTag(key));
            original.put(key, nested);
        }
        QuestProgress.copyTo(original, replacement);
        QuestCooldownStore.copyTo(original, replacement);
        for (String key : keys) {
            check(replacement.contains(key) && replacement.get(key) != original.get(key), "Respawn copies every quest field without aliasing: " + key);
            CompoundTag copied = (CompoundTag) replacement.get(key);
            copied.put("later", new ValueTag("changed"));
            check(!((CompoundTag) original.get(key)).contains("later"), "Replacement quest state must not mutate the old player");
        }
        QuestProgress.copyTo(new CompoundTag(), replacement);
        QuestCooldownStore.copyTo(new CompoundTag(), replacement);
        for (String key : keys) check(!replacement.contains(key), "Missing source fields clear stale replacement state");
        check(replacement.contains("Unrelated"), "Copying quest state preserves other player data");
    }

    private static void connectionCleanup() {
        for (boolean login : new boolean[]{false, true}) {
            ClientQuestState.setQuestState(8, Map.of(3, 5000L));
            ClientMemoryDigest.accept("previous world's hero memory");
            ClientAgentStatusStore.accept(new AgentStatusSnapshot("old server"));
            CompletableFuture<ClientAgentToolRequestStore.AgentToolResultRecord> pending = new CompletableFuture<>();
            ClientAgentToolRequestStore.track(UUID.randomUUID(), pending);
            ClientAgentToolRequestStore.complete(UUID.randomUUID(), "old_tool", true, "old result");
            if (login) ClientForgeEvents.onClientPlayerLogin(new ClientPlayerNetworkEvent.LoggingIn());
            else ClientForgeEvents.onClientPlayerLogout(new ClientPlayerNetworkEvent.LoggingOut());
            check(ClientQuestState.activeQuestId() == 0 && ClientQuestState.cooldownEnd(3) == 0, "Connection transitions clear old quests and cooldowns");
            check(ClientMemoryDigest.get().isEmpty(), "Old hero memory is never injected into the next world's chat");
            check(!ClientAgentStatusStore.hasData() && ClientAgentStatusStore.ageSeconds() == -1, "Old agent status is not presented as current");
            check(ClientAgentToolRequestStore.pendingCount() == 0 && pending.isDone() && pending.join() == null,
                    "Disconnect/login releases pending tool requests instead of carrying them into the next server");
            check(ClientAgentToolRequestStore.recentResult() == null, "Old tool results are cleared");
        }
        PacketHandler.requests.clear();
        ClientMemoryDigest.ensureFresh();
        check(PacketHandler.requests.size() == 1 && PacketHandler.requests.get(0) instanceof RequestHeroMemoryDigestPacket,
                "The new session requests a fresh memory digest immediately");
    }

    private static void liveQuestCooldown() {
        HeroRequestScreen screen = new HeroRequestScreen();
        ClientQuestState.setQuestState(0, Map.of(3, 10L));
        screen.now = 9;
        screen.refreshAcceptState();
        check(!screen.acceptButton.active, "An unexpired cooldown disables acceptance");
        screen.now = 10;
        screen.refreshAcceptState();
        check(screen.acceptButton.active, "The open screen enables acceptance when the cooldown expires");
        ClientQuestState.setQuestState(8, Map.of());
        screen.refreshAcceptState();
        check(!screen.acceptButton.active, "An active quest prevents starting another quest");
    }
