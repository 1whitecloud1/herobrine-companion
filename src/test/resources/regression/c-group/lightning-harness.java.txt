    public java.util.List<Object> run(int variant, Vec3 start, Vec3 finish, double width) {
        this.kind = variant == 0 ? Kind.ARC : Kind.BEAM;
        this.radius = Math.max(variant == 0 ? 1.5D : 2.5D, width);
        this.origin = start; this.end = finish; this.level = new FakeLevel();
        this.caster = new LivingEntity("caster",new Vec3(5,-1,0));
        level.targets.add(caster);
        level.targets.add(new LivingEntity("center",new Vec3(5,-1,0)));
        level.targets.add(new LivingEntity("edge",new Vec3(5,-1,2)));
        level.targets.add(new LivingEntity("outside",new Vec3(5,-1,8)));
        level.targets.add(new LivingEntity("beforeStart",new Vec3(-2,-1,0)));
        level.targets.add(new HeroEntity("hero",new Vec3(5,-1,0)));
        java.util.List<Object> result = new java.util.ArrayList<>();
        result.add(doTick(new TickBudget()));
        // A retry must not re-emit the already prepared sound/damage stage.
        result.add(doTick(new TickBudget()));
        result.addAll(level.events);
        result.add(level.cleared);
        for(LivingEntity entity:level.targets) result.add(entity.name+":"+entity.damage+":"+entity.push);
        return result;
    }
