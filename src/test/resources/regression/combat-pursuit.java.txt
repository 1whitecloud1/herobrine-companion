import java.util.*;

public class CombatPursuitProbe {
    private static int checks;
    public static void verify() {
        stableCruiseAndLanding();
        pursuitYieldsToAttacks();
        groundProgressAndFlightCooldown();
        airborneComboContinuation();
        waterExitDoesNotCancelLanding();
        System.out.println("CombatPursuitChecks: " + checks + " checks passed");
    }

    private static void stableCruiseAndLanding() {
        HeroEntity hero = new HeroEntity(0, 64, 0);
        LivingEntity target = new LivingEntity(20, 64, 0);
        for (double height : new double[]{64, 66, 72, 100}) {
            hero.y = height;
            check(HeroCombatPursuit.flightTarget(hero, target, 3).y == 66,
                    "Open-ground cruise must not climb with Hero's current height");
        }
        hero.world.surface = 70;
        check(HeroCombatPursuit.flightTarget(hero, target, 3).y == 72, "Cruise still clears an obstacle");
        target.x = 1;
        for (double surface : new double[]{64, -1, 0, -32, -63, 64.5}) {
            hero.world.surface = surface;
            target.y = surface;
            for (double hover : new double[]{0, .05, .5, 2, 5}) {
                hero.y = surface + hover;
                check(HeroCombatPursuit.shouldLand(hero, target, 3),
                        "Reachable ground, including negative Y and slabs, must allow landing");
            }
        }
        hero.y = 70;
        hero.world.surface = 64;
        target.y = 64;
        check(HeroCombatPursuit.shouldLand(hero, target, 3), "A six-block descent must restore ground combat");
        hero.y = 71;
        check(!HeroCombatPursuit.shouldLand(hero, target, 3), "A drop beyond landing depth must keep flight");
        hero.y = 65;
        hero.world.headroom = false;
        check(!HeroCombatPursuit.shouldLand(hero, target, 3), "Landing needs clearance for Hero's entire body");
        hero.world.headroom = true;
        hero.world.fluid = true;
        check(!HeroCombatPursuit.shouldLand(hero, target, 3), "A fluid landing must be rejected");
        hero.world.fluid = false;
        target.y = 68;
        check(!HeroCombatPursuit.shouldLand(hero, target, 3), "Ground too low to reach a high target must be rejected");
        hero.water = true;
        check(HeroCombatPursuit.shouldTakeFlight(hero, target, 0), "Fluid escape must still take flight");
        hero.water = false;
        target.y = 69;
        check(HeroCombatPursuit.shouldTakeFlight(hero, target, 0), "High targets must still take flight");
        hero.passenger = true;
        check(!HeroCombatPursuit.shouldTakeFlight(hero, target, 3), "Riding must not start pursuit flight");
    }

    private static void pursuitYieldsToAttacks() {
        HeroEntity hero = new HeroEntity(0, 64, 0);
        hero.target = new LivingEntity(20, 70, 0);
        HeroEpicFightChaseGoal goal = new HeroEpicFightChaseGoal(hero, 1, 3);
        goal.start();
        hero.midAttack = true;
        Vec3 attackMotion = new Vec3(.5, .62, .1);
        hero.velocity = attackMotion;
        hero.control.moving = true;
        check(!goal.canUse(), "An attack must not start a new chase goal");
        for (int tick = 0; tick < 80; tick++) {
            hero.tickCount++;
            goal.tick();
            check(!hero.floating && hero.velocity == attackMotion && !hero.control.moving,
                    "Pursuit must preserve attack root motion without taking flight");
        }
        hero.midAttack = false;
        goal.tick();
        check(hero.floating && hero.noGravity, "A high target is chased after the attack finishes");
        hero.midAttack = true;
        hero.target = new LivingEntity(1, 64, 0);
        hero.y = 65;
        hero.velocity = attackMotion;
        goal.tick();
        HeroCombatPursuit.landHeroIfCombatIdle(hero, hero.target, 3, true);
        check(hero.floating && hero.velocity == attackMotion, "Landing must not interrupt an aerial attack");
        hero.midAttack = false;
        goal.tick();
        check(!hero.floating && !hero.noGravity && !hero.control.moving && hero.velocity.y < 0,
                "A safe melee landing must immediately cancel the flight movement command");
    }

    private static void groundProgressAndFlightCooldown() {
        HeroEntity hero = new HeroEntity(0, 64, 0);
        hero.target = new LivingEntity(50, 64, 0);
        hero.navigation.pathWorks = false;
        HeroEpicFightChaseGoal goal = new HeroEpicFightChaseGoal(hero, 1, 3);
        goal.start();
        check(goal.requiresUpdateEveryTick(), "Flight timers must run on server ticks");
        for (int tick = 0; tick < 100; tick++) {
            hero.tickCount++;
            hero.x += .1;
            goal.tick();
            check(!hero.floating, "Failed path planning while making ground progress must not cause flight");
        }
        for (int tick = 0; tick < 60; tick++) { hero.tickCount++; goal.tick(); }
        check(hero.floating, "A genuinely stalled ground chase must retain flight escape");
        hero.target.x = hero.x + 1;
        hero.y = 65;
        hero.tickCount++;
        goal.tick();
        check(!hero.floating, "Arriving at a ground target must restore gravity");
        goal.stop();
        goal.start();
        hero.target.x = hero.x + 20;
        for (int tick = 0; tick < 45; tick++) {
            hero.tickCount++;
            goal.tick();
            check(!hero.floating, "Landing cooldown must survive a goal restart");
        }
        for (int tick = 0; tick < 30; tick++) { hero.tickCount++; goal.tick(); }
        check(hero.floating, "A persistent obstacle may take flight after the landing cooldown");

        hero = new HeroEntity(0, 64, 0);
        hero.target = new LivingEntity(4, 64, 0);
        hero.navigation.pathWorks = false;
        goal = new HeroEpicFightChaseGoal(hero, 1, 3);
        goal.start();
        for (int tick = 0; tick < 100; tick++) { hero.tickCount++; goal.tick(); }
        check(!hero.floating, "Near-melee navigation failures must not starve ground combos");
        hero.target.x = 1;
        hero.sight = false;
        check(goal.canUse(), "A nearby target behind a wall still needs a chase");
        for (int tick = 0; tick < 60; tick++) { hero.tickCount++; goal.tick(); }
        check(hero.floating, "A nearby wall must still permit flight escape");
        check(!HeroCombatPursuit.shouldLand(hero, hero.target, 3), "Do not land before clearing the wall");
        hero.sight = true;
        goal.tick();
        check(!hero.floating, "Land once the wall is cleared and melee is reachable");
    }

    private static void airborneComboContinuation() {
        HeroEntity hero = new HeroEntity(0, 64, 0);
        hero.target = new LivingEntity(1, 64, 0);
        HumanoidMobPatch<?> patch = new HumanoidMobPatch<>(hero);
        HeroNightfallProfile profile = new HeroNightfallProfile();
        check(Nightfall.canStartDefaultCombo(patch, profile, 0, 3.6), "Nightfall ground combo can open");
        hero.grounded = false;
        check(!Nightfall.canStartDefaultCombo(patch, profile, 0, 3.6), "Nightfall opener still needs ground contact");
        for (int segment = 1; segment < 6; segment++) {
            check(Nightfall.canContinueDefaultCombo(patch, profile, segment, 3.6),
                    "Nightfall authored jumps must not discard later combo segments");
        }
        HeroEpicFightPatch playerLike = new HeroEpicFightPatch();
        PlayerLikeAttackProfile playerProfile = new PlayerLikeAttackProfile();
        HeroCombatPlanner.preferCombo = false;
        check(!playerLike.canStartPlayerLikeGroundCombo(patch, playerProfile, 0), "Epic Fight opener still needs ground contact");
        for (int segment = 1; segment < 3; segment++) {
            hero.combo = segment;
            hero.velocity = new Vec3(.5, .2, 0);
            check(playerLike.canStartPlayerLikeGroundCombo(patch, playerProfile, segment),
                    "Epic Fight continuation must tolerate airborne root motion and a changed utility preference");
        }
        hero.hold = true;
        check(!Nightfall.canContinueDefaultCombo(patch, profile, 1, 3.6)
                && !playerLike.canStartPlayerLikeGroundCombo(patch, playerProfile, 2), "Hold actions still exclude basic combo continuation");
        hero.hold = false;
        hero.battle = false;
        check(!Nightfall.canContinueDefaultCombo(patch, profile, 1, 3.6)
                && !playerLike.canStartPlayerLikeGroundCombo(patch, playerProfile, 2), "Leaving combat still excludes continuation");
    }

    private static void waterExitDoesNotCancelLanding() {
        HeroEntity hero = new HeroEntity(0, 64, 0);
        hero.target = new LivingEntity(1, 64, 0);
        HeroMovementState movement = new HeroMovementState(hero);
        hero.water = true;
        movement.afterAiStep();
        check(hero.floating && hero.noGravity, "Combat must escape water");
        hero.water = false;
        HeroCombatPursuit.land(hero);
        for (int tick = 0; tick < 40; tick++) {
            movement.afterAiStep();
            check(!hero.floating && !hero.noGravity, "Water-exit grace must not undo combat landing");
        }
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
        checks++;
    }

    static final class HeroCombatPursuit { /* PURSUIT */ }

        static LivingEntity getTrackedTarget(HeroEntity hero) { return hero == null ? null : hero.target; }

    }
    static class HeroEpicFightPatch {

        LivingEntity getTrackedTarget(HeroEntity hero) { return hero.target; }
        HeroCombatPlanner.CombatTuning getPlayerLikeCombatTuning(PlayerLikeAttackProfile profile) { return null; }
        boolean isDashLikeMovement(HeroEntity hero, PlayerLikeAttackProfile profile) { return hero.velocity.horizontalDistanceSqr() > .02; }

    }
    static final class HeroMovementState {

    }
    static class Goal {

    static class LivingEntity {

        final Level world = new Level();

        double horizontalDistanceSqr() { return x * x + z * z; }
